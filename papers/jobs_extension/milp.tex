% ==============================================================================
\section{MILP Formulation}
\label{sec:milp}
% ==============================================================================

We present a mixed-integer linear programming formulation of the
problem of \S\ref{sec:problem}.  Where possible the notation follows
the conventions used elsewhere in the literature for aircraft
positioning problems with blocking arcs: parameters use upper-case
Latin letters, decision variables use lower-case Latin letters, and
Greek letters are reserved for fixed scalar constants ($\varepsilon$,
$\mu$, $\delta$, the time-granularity $\eta$ and the big-$M$
relaxation~$M$).  The key feature of the formulation is that every
access of a blocked aircraft is explicitly \emph{classified}, by means
of a partition constraint, into one of the four mutually exclusive
states allowed by the problem: before the front aircraft's stay,
after it, in an inter-job gap (Mode~B), or strictly inside an
interruptible job (Mode~C).  An access that cannot be classified into
any of these four states corresponds to an attempt to enter or leave a
position whose blocker is in the middle of a non-interruptible job,
and is correctly rendered infeasible by the model.

\subsection*{Sets}

\begin{tabular}{@{}l@{\quad}p{0.78\textwidth}@{}}
  $\calP$
    & maintenance positions \\
  $\calR$
    & aircraft \\
  $\calJ_r,\ r \in \calR$
    & ordered job chain of aircraft $r$;
      $\calJ_r = (j^r_1, \ldots, j^r_{N_r})$ \\
  $\calJ = \bigcup_{r \in \calR} \calJ_r$
    & global job set \\
  $\calA \subseteq \calP \times \calP$
    & blocking arcs $(p, p')$: position $p$ blocks position $p'$ \\
\end{tabular}

\subsection*{Parameters}

\begin{tabular}{@{}l@{\quad}p{0.78\textwidth}@{}}
  $D_j$
    & nominal duration of job $j \in \calJ$ \\
  $I_j \in \{0, 1\}$
    & interruptibility flag of job $j$; $I_j = 1$ iff $j$ can be paused \\
  $E_r, L_r$
    & earliest start and target finish of aircraft $r$ \\
  $\varepsilon$
    & tow-in / tow-out time; minimum separation between consecutive
      aircraft serviced at the same position \\
  $\mu$
    & operational time absorbed by a single Mode~B manoeuvre inside
      an inter-job gap (per-access, cumulative across all manoeuvres
      that share the same gap) \\
  $\delta$
    & additional working time added to an interruptible job each time
      it is paused to allow a Mode~C manoeuvre \\
  $\eta$
    & time-granularity used to enforce strict inequalities on the
      Mode~C job interior; equals the smallest time difference
      meaningful in the instance (e.g.\ one day) \\
  $K^{\max}_j$
    & valid upper bound on the number of times job $j$ may be paused;
      we take $K^{\max}_j = 2 (|\calR| - 1)$ \\
  $H^{UB}$
    & technical upper bound on every job start and finish time,
      large enough to accommodate the worst-case sequential schedule
      with all possible interruptions \\
  $M$
    & big-$M$ constant; we take $M = H^{UB} + \max(\varepsilon, \mu, \delta, \eta)$ \\
  $W^M, W^D, W^S$
    & weights for makespan, total delay, and movements \\
\end{tabular}

\subsection*{Variables}

\fmlgroup{Assignment and timing}

\begin{tabular}{@{}l@{\quad}p{0.78\textwidth}@{}}
  $x_{rp} \in \{0,1\}$
    & $1$ iff aircraft $r$ is assigned to position $p$;
      $\forall r \in \calR,\ p \in \calP$ \\
  $q_{rr'p} \in \{0,1\}$
    & $1$ iff aircraft $r$ precedes aircraft $r'$ at position $p$
      (both assigned to $p$);
      $\forall r \ne r' \in \calR,\ p \in \calP$ \\
  $s_j, f_j \in [0, H^{UB}]$
    & start and finish time of job $j$;
      $\forall j \in \calJ$ \\
  $\kappa_j \in \{0, \ldots, K^{\max}_j\}$
    & number of times job $j$ is paused (Mode~C interruption counter);
      $\forall j \in \calJ$ \\
  $s_r, f_r$
    & aircraft start / finish shortcuts,
      $s_r = s_{j^r_1}$ and $f_r = f_{j^r_{N_r}}$;
      $\forall r \in \calR$ \\
\end{tabular}

\fmlgroup{Placement compatibility}

\begin{tabular}{@{}l@{\quad}p{0.78\textwidth}@{}}
  $y_{rr'pp'} \in \{0,1\}$
    & $1$ iff $r$ is the front aircraft at $p$ and $r'$ is the rear
      aircraft at $p'$;
      $\forall r \ne r' \in \calR,\ (p, p') \in \calA$ \\
\end{tabular}

\fmlgroup{Access classification}
For every $r \ne r' \in \calR$, every $(p, p') \in \calA$, and every
access side $a \in \{\mathrm{in}, \mathrm{out}\}$ we use the convention
$\tau^a_{r'} = s_{r'}$ when $a = \mathrm{in}$ and $\tau^a_{r'} = f_{r'}$
when $a = \mathrm{out}$, and introduce four mutually exclusive
classification indicators:

\begin{tabular}{@{}l@{\quad}p{0.78\textwidth}@{}}
  $z^{-,a}_{rr'pp'} \in \{0,1\}$
    & $1$ iff access $\tau^a_{r'}$ falls strictly before $r$'s stay \\
  $z^{+,a}_{rr'pp'} \in \{0,1\}$
    & $1$ iff access $\tau^a_{r'}$ falls strictly after $r$'s stay \\
  $b^{Ba}_{rr'pp'k} \in \{0,1\}$
    & $1$ iff access $\tau^a_{r'}$ falls in inter-job gap $k$ of $r$
      (Mode B); $\forall k = 1, \ldots, N_r-1$ \\
  $b^{Ca}_{rr'pp'j} \in \{0,1\}$
    & $1$ iff access $\tau^a_{r'}$ falls strictly inside job $j$ of $r$
      (Mode C); $\forall j \in \calJ_r$ \\
\end{tabular}

\fmlgroup{Objective auxiliaries}

\begin{tabular}{@{}l@{\quad}p{0.78\textwidth}@{}}
  $m \in \NR$
    & makespan (latest aircraft finish time) \\
  $v^D_r \in \NR$
    & positive delay of aircraft $r$ past its target finish $L_r$;
      $\forall r \in \calR$ \\
  $n \in \NZ$
    & total number of manoeuvres (movement count) \\
\end{tabular}

\subsection*{Constraints}

To keep the alignment compact we abbreviate
$\forall r \ne r' \in \calR,\ (p,p') \in \calA,\ a \in
\{\mathrm{in},\mathrm{out}\}$ by $(\dagger)$ and
$\forall r \ne r' \in \calR,\ p \in \calP$ by $(\ddagger)$.

\fmlgroup{Assignment, chain, earliest start, sequencing}
\begin{align}
  \sum_{p \in \calP} x_{rp} &= 1
    && \forall r \in \calR \label{eq:assign}\\
  f_j &= s_j + D_j + \delta\, \kappa_j
    && \forall j \in \calJ \label{eq:jobdur}\\
  s_{j^r_{k+1}} &\ge f_{j^r_k}
    && \forall r,\ k = 1, \ldots, N_r{-}1 \label{eq:chain}\\
  s_r &\ge E_r
    && \forall r \in \calR \label{eq:earlystart}\\
  s_{r'} &\ge f_r + \varepsilon - M(1 - q_{rr'p}) \notag\\
         &\quad - M(1 - x_{rp}) - M(1 - x_{r'p})
    && (\ddagger) \label{eq:sep}\\
  q_{rr'p} + q_{r'rp} &\ge x_{rp} + x_{r'p} - 1
    && \forall r < r' \in \calR,\, p \in \calP \label{eq:ordLB}\\
  q_{rr'p} + q_{r'rp} &\le 1
    && \forall r < r' \in \calR,\, p \in \calP \label{eq:ordUB}\\
  q_{rr'p} &\le x_{rp},\quad q_{rr'p} \le x_{r'p}
    && (\ddagger),\ \forall p \in \calP \label{eq:ordTight}
\end{align}
Constraint~\eqref{eq:assign} assigns each aircraft to exactly one
position.  Equation~\eqref{eq:jobdur} fixes each job's finish as its
start plus its nominal duration plus the extension $\delta\kappa_j$
accrued by its interruptions.  Constraint~\eqref{eq:chain} enforces the
job-chain precedence inside an aircraft, which, together with the
shortcuts $s_r = s_{j^r_1}$ and $f_r = f_{j^r_{N_r}}$, chains its jobs
end-to-end.  Constraint~\eqref{eq:earlystart} forbids any aircraft from
starting before its earliest start $E_r$.  The big-$M$
disjunction~\eqref{eq:sep} imposes the $\varepsilon$ separation between
two aircraft that share a position when $r$ precedes $r'$ there, and is
vacuous otherwise.  Constraints~\eqref{eq:ordLB}
and~\eqref{eq:ordUB} make the ordering binaries consistent --- at least
one direction is chosen whenever both aircraft share the position, and
at most one is --- while \eqref{eq:ordTight} ties each $q_{rr'p}$ to the
two assignments it presupposes.

\fmlgroup{Placement compatibility}
For every $r \ne r' \in \calR$ and every $(p, p') \in \calA$:
\begin{align}
  y_{rr'pp'} &\le x_{rp},\quad
  y_{rr'pp'} \le x_{r'p'},\quad
  y_{rr'pp'} \ge x_{rp} + x_{r'p'} - 1
    \label{eq:y_link}
\end{align}
Inequalities~\eqref{eq:y_link} make $y_{rr'pp'}$ the logical conjunction
of $x_{rp}$ and $x_{r'p'}$: the first two force $y_{rr'pp'} = 0$ unless
both placements hold, and the third forces $y_{rr'pp'} = 1$ when they
do, so $y_{rr'pp'} = 1$ exactly when the blocking interaction of the
arc $(p, p')$ is materialised by the assignment.

\fmlgroup{Exhaustive classification of every access}
For every $(\dagger)$, an access $\tau^a_{r'}$ that is exposed to the
front aircraft $r$ must be classified into exactly one of the four
allowed states; conversely, if $y_{rr'pp'} = 0$ the access is not
exposed and no indicator may fire:
\begin{align}
  z^{-,a}_{rr'pp'} + z^{+,a}_{rr'pp'}
   + \!\!\sum_{k=1}^{N_r-1}\! b^{Ba}_{rr'pp'k}
   + \!\!\sum_{j \in \calJ_r}\! b^{Ca}_{rr'pp'j}
   &= y_{rr'pp'}
    && \label{eq:partition}
\end{align}
The partition constraint~\eqref{eq:partition} is the structural core of
the model.  When the blocking interaction is materialised
($y_{rr'pp'} = 1$) it forces the access $\tau^a_{r'}$ into exactly one
of the four admissible states; when it is not ($y_{rr'pp'} = 0$) it
forbids every indicator from firing.  An access that cannot satisfy any
of the four state definitions below --- because its blocker is mid-way
through a non-interruptible job --- leaves~\eqref{eq:partition}
unsatisfiable, so the only feasible response is to revise the timing or
the assignment; this is what makes the model exact.

\fmlgroup{Before / after the front stay}
For every $(\dagger)$:
\begin{align}
  \tau^a_{r'} &\le s_r - \eta + M\bigl(1 - z^{-,a}_{rr'pp'}\bigr)
    && \label{eq:zminus}\\
  \tau^a_{r'} &\ge f_r + \eta - M\bigl(1 - z^{+,a}_{rr'pp'}\bigr)
    && \label{eq:zplus}
\end{align}
Constraint~\eqref{eq:zminus} is active only when
$z^{-,a}_{rr'pp'} = 1$, where it places the access at least $\eta$
before the front aircraft's start; symmetrically,
\eqref{eq:zplus} places the access at least $\eta$ after the front's
finish when $z^{+,a}_{rr'pp'} = 1$.  In both cases the front position is
vacant at the access instant (Mode~A, no manoeuvre), and the big-$M$
term deactivates the bound for the state that is not selected.

A remark on $\eta$ is in order: it is an instance datum, not a solver
tolerance.  On the benchmark of \S\ref{sec:experiments} it equals one full
day --- the grid on which all durations and time windows are drawn --- so
the strict-inequality margins it enforces sit many orders of magnitude
above the solver's feasibility and integrality tolerances, and we observed
no numerical sensitivity to it.  On instances with a finer time grid,
$\eta$ should scale with that grid rather than be set to an arbitrarily
small constant, which would indeed risk tolerance-level artefacts.

\fmlgroup{Mode B --- inter-job gap}
For every $(\dagger)$ and every $k = 1, \ldots, N_r-1$:
\begin{align}
  \tau^a_{r'} &\ge f_{j^r_k} - M\bigl(1 - b^{Ba}_{rr'pp'k}\bigr)
    && \label{eq:bb_lb}\\
  \tau^a_{r'} &\le s_{j^r_{k+1}} + M\bigl(1 - b^{Ba}_{rr'pp'k}\bigr)
    && \label{eq:bb_ub}
\end{align}
Each Mode~B event consumes $\mu$ time units of the gap, so the gap
must be wide enough to accommodate all events that use it:
\begin{align}
  s_{j^r_{k+1}} - f_{j^r_k}
    &\ge
    \mu \!\!\sum_{\substack{r' \ne r\\ (p,p') \in \calA \\ a \in \{\mathrm{in}, \mathrm{out}\}}}
        \!\! b^{Ba}_{rr'pp'k}
    && \forall r \in \calR,\ k = 1, \ldots, N_r-1
    \label{eq:bb_gap}
\end{align}
The bounds~\eqref{eq:bb_lb} and~\eqref{eq:bb_ub} are active when
$b^{Ba}_{rr'pp'k} = 1$ and then confine the access to inter-job gap $k$
of the front aircraft, between the finish of $j^r_k$ and the start of
$j^r_{k+1}$; \eqref{eq:bb_gap} requires that gap to be wide enough to
absorb $\mu$ time units for every Mode~B access routed through it.

\fmlgroup{Mode C --- strictly inside an interruptible job}
For every $(\dagger)$ and every $j \in \calJ_r$:
\begin{align}
  \tau^a_{r'} &\ge s_j + \eta - M\bigl(1 - b^{Ca}_{rr'pp'j}\bigr)
    && \label{eq:bc_lb}\\
  \tau^a_{r'} &\le s_j + D_j + \delta\bigl(\kappa_j - b^{Ca}_{rr'pp'j}\bigr)
                - \eta + M\bigl(1 - b^{Ca}_{rr'pp'j}\bigr)
    && \label{eq:bc_ub}\\
  b^{Ca}_{rr'pp'j} &\le I_j
    && \label{eq:bc_interr}
\end{align}
The lower bound~\eqref{eq:bc_lb} is active when
$b^{Ca}_{rr'pp'j} = 1$ and forces the access strictly inside job $j$,
at least $\eta$ after its start.  The upper bound~\eqref{eq:bc_ub} pins
$\tau^a_{r'}$ below the job's \emph{active} finish, excluding the
extension that this specific access would itself generate, and so
prevents the ``self-justifying interruption'' phenomenon in which an
access could otherwise become feasible only because the very
interruption it triggers extends the job past it.  This correction
deserves emphasis: the processing times of this model are
\emph{endogenous} --- through $\kappa_j$ they depend on the very
classification decisions the constraints govern --- and a bound written
against the full extended finish would let an access bootstrap its own
feasibility.  Excluding the access's own contribution from the bound is
the guard that any formulation with decision-dependent durations needs
at this point.
Constraint~\eqref{eq:bc_interr} forbids Mode~C on non-interruptible
jobs ($I_j = 0$).

\fmlgroup{Counters, delay and makespan}
\begin{align}
  \kappa_j &=
    \!\!\sum_{\substack{r' \ne r\\ (p,p') \in \calA\\ a \in \{\mathrm{in}, \mathrm{out}\}}}
    \!\! b^{Ca}_{rr'pp'j}
    && \forall r \in \calR,\ j \in \calJ_r \label{eq:kappa}\\
  n &= 2 \!\!\sum_{(p,p') \in \calA}
       \!\!\sum_{\substack{r,r' \in \calR \\ r \ne r'}}
       \sum_{a}
       \biggl(
         \!\sum_{k=1}^{N_r-1}\! b^{Ba}_{rr'pp'k}
         + \!\sum_{j \in \calJ_r}\! b^{Ca}_{rr'pp'j}
       \biggr)
    && \label{eq:movements}\\
  v^D_r &\ge f_r - L_r
    && \forall r \in \calR \label{eq:delay}\\
  m &\ge f_r
    && \forall r \in \calR \label{eq:makespan}
\end{align}
Equation~\eqref{eq:kappa} aggregates the Mode~C indicators of job $j$
into its interruption counter $\kappa_j$, closing the loop with the
duration constraint~\eqref{eq:jobdur}.  Equation~\eqref{eq:movements}
totals the manoeuvre count over all Mode~B and Mode~C events, each
taken with the factor of two for the tow-out and tow-back displacement.
Constraint~\eqref{eq:delay} linearises the positive delay
$v^D_r = \max(0,\, f_r - L_r)$ --- the lower bound combined with the
$v^D_r \ge 0$ domain and the minimisation direction --- and
\eqref{eq:makespan} sets the makespan $m$ to the latest aircraft
finish.

All disjunctions share the single constant
$M = H^{UB} + \max(\varepsilon, \mu, \delta, \eta)$.  This uniform choice
is valid but not the tightest possible: pair-dependent constants
$M_{rr'pp'}$, derived from the projected time windows of the two aircraft
involved (their earliest starts, chain lengths and interruption bounds),
would strengthen the linear relaxation at no modelling cost.  We retain
the uniform constant for readability and have not explored this
tightening; it is a natural refinement alongside the valid inequalities
discussed in \S\ref{sec:conclusions}.

\subsection*{Objective}
\begin{equation}
  \min \quad W^M\, m
         \;+\; W^D \sum_{r \in \calR} v^D_r
         \;+\; W^S\, n.
  \label{eq:milpobj}
\end{equation}
The objective~\eqref{eq:milpobj} minimises the weighted sum of the
makespan, the total positive delay past the target finishes, and the
movement count, under non-negative user weights $W^M$, $W^D$, $W^S$;
the weight profiles used in the experiments are discussed in
\S\ref{sec:exp_setup}.
