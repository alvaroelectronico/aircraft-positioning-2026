% ==============================================================================
\section{MILP formulation}
\label{sec:milp}
% ==============================================================================

We present a mixed-integer linear programming formulation of the
problem of \S\ref{sec:problem}. 
% Where possible the notation follows
% the conventions used elsewhere in the literature for aircraft
% positioning problems with blocking arcs: parameters use upper-case
% Latin letters, decision variables use lower-case Latin letters, and
% Greek letters are reserved for fixed scalar constants ($\varepsilon$,
% $\mu$, $\delta$, the time-granularity $\eta$ and the big-$M$
% relaxation~$M$). 
Its central element is a partition constraint that classifies every
access of a blocked aircraft into one of the four mutually exclusive
states allowed by the problem: before the front aircraft's stay, after
it, in an inter-job gap (Mode~B), or strictly inside an interruptible
job (Mode~C); the constraint is discussed with
equation~\eqref{eq:partition}.

The formulation retains the sets and parameters of \S\ref{sec:problem}
and introduces four constants: the time granularity $\eta$, the smallest
time difference meaningful in the instance (one day on the benchmark of
\S\ref{sec:experiments}), used to enforce strict inequalities on the
interior of Mode~C jobs; a valid bound $K^{\max}_j = 2(|\calR| - 1)$ on
the number of times job $j$ may be paused; a technical upper bound
$H^{UB}$ on every job start and finish time, large enough to accommodate
the worst-case sequential schedule with all possible interruptions; and
the big-$M$ constant $M = H^{UB} + \max(\varepsilon, \mu, \delta, \eta)$.

\subsection*{Variables}

\fmlgroup{Assignment and timing}

\begin{tabular}{@{}l@{\quad}p{0.78\textwidth}@{}}
  $x_{rp} \in \{0,1\}$
    & $1$ iff aircraft $r$ is assigned to position $p$;
      $\forall r \in \calR,\ p \in \calP$, 0 otherwise \\
  $q_{rr'p} \in \{0,1\}$
    & $1$ iff aircraft $r$ precedes aircraft $r'$ at position $p$
      (both assigned to $p$);
      $\forall r \ne r' \in \calR,\ p \in \calP$, 0 otherwise\\
  $s_j, f_j \in [0, H^{UB}]$
    & start and finish time of job $j$;
      $\forall j \in \calJ$ \\
  $\kappa_j \in \{0, \ldots, K^{\max}_j\}$
    & number of times job $j$ is paused (Mode~C interruption counter);
      $\forall j \in \calJ$ \\
  $s_r, f_r$
    & aircraft start / finish shortcuts,
      $s_r = s_{j^r_1}$ and $f_r = f_{j^r_{N_r}}$;
      $\forall r \in \calR$ \\
\end{tabular}

\fmlgroup{Placement compatibility}

\begin{tabular}{@{}l@{\quad}p{0.78\textwidth}@{}}
  $y_{rr'pp'} \in \{0,1\}$
    & $1$ iff $r$ is the front aircraft at $p$ and $r'$ is the rear
      aircraft at $p'$;
      $\forall r \ne r' \in \calR,\ (p, p') \in \calA$, 0 otherwise \\
\end{tabular}

\fmlgroup{Access classification}
For every $r \ne r' \in \calR$, every $(p, p') \in \calA$, and every
access side $a \in \{\mathrm{in}, \mathrm{out}\}$ we use the convention
$\tau^a_{r'} = s_{r'}$ when $a = \mathrm{in}$ and $\tau^a_{r'} = f_{r'}$
when $a = \mathrm{out}$, and introduce four mutually exclusive
classification indicators:

\begin{tabular}{@{}l@{\quad}p{0.78\textwidth}@{}}
  $z^{-,a}_{rr'pp'} \in \{0,1\}$
    & $1$ iff access $\tau^a_{r'}$ falls strictly before $r$'s stay, 0 otherwise \\
  $z^{+,a}_{rr'pp'} \in \{0,1\}$
    & $1$ iff access $\tau^a_{r'}$ falls strictly after $r$'s stay, 0 otherwise \\
  $b^{Ba}_{rr'pp'k} \in \{0,1\}$
    & $1$ iff access $\tau^a_{r'}$ falls in inter-job gap $k$ of $r$
      (Mode B), 0 otherwise; $\forall k = 1, \ldots, N_r-1$ \\
  $b^{Ca}_{rr'pp'j} \in \{0,1\}$
    & $1$ iff access $\tau^a_{r'}$ falls strictly inside job $j$ of $r$
      (Mode C), 0 otherwise; $\forall j \in \calJ_r$ \\
\end{tabular}

\fmlgroup{Objective auxiliaries}

\begin{tabular}{@{}l@{\quad}p{0.78\textwidth}@{}}
  $m \in \NR$
    & makespan (latest aircraft finish time) \\
  $v^D_r \in \NR$
    & positive delay of aircraft $r$ past its target finish $L_r$;
      $\forall r \in \calR$ \\
  $n \in \NZ$
    & total number of towing movements (two per access manoeuvre) \\
\end{tabular}

\subsection*{Constraints}

To keep the alignment compact we abbreviate
$\forall r \ne r' \in \calR,\ (p,p') \in \calA,\ a \in
\{\mathrm{in},\mathrm{out}\}$ by $(\dagger)$ and
$\forall r \ne r' \in \calR,\ p \in \calP$ by $(\ddagger)$.

\fmlgroup{Assignment, chain, earliest start, sequencing}
\begin{align}
  \sum_{p \in \calP} x_{rp} &= 1
    && \forall r \in \calR \label{eq:assign}\\
  f_j &= s_j + D_j + \delta\, \kappa_j
    && \forall j \in \calJ \label{eq:jobdur}\\
  s_{j^r_{k+1}} &\ge f_{j^r_k}
    && \forall r,\ k = 1, \ldots, N_r{-}1 \label{eq:chain}\\
  s_r &\ge E_r
    && \forall r \in \calR \label{eq:earlystart}\\
  s_{r'} &\ge f_r + \varepsilon - M(1 - q_{rr'p}) \notag\\
         &\quad - M(1 - x_{rp}) - M(1 - x_{r'p})
    && (\ddagger) \label{eq:sep}\\
  q_{rr'p} + q_{r'rp} &\ge x_{rp} + x_{r'p} - 1
    && \forall r < r' \in \calR,\, p \in \calP \label{eq:ordLB}\\
  q_{rr'p} + q_{r'rp} &\le 1
    && \forall r < r' \in \calR,\, p \in \calP \label{eq:ordUB}\\
  q_{rr'p} &\le x_{rp},\quad q_{rr'p} \le x_{r'p}
    && (\ddagger),\ \forall p \in \calP \label{eq:ordTight}
\end{align}
Constraint~\eqref{eq:assign} assigns each aircraft to one position;
\eqref{eq:jobdur} fixes each job's finish as its start plus its nominal
duration plus the extension $\delta\kappa_j$ accrued by its
interruptions; \eqref{eq:chain} and~\eqref{eq:earlystart} enforce the
job chain, with $s_r = s_{j^r_1}$ and $f_r = f_{j^r_{N_r}}$, and the
earliest start; and \eqref{eq:sep}--\eqref{eq:ordTight} sequence the
aircraft that share a position with the $\varepsilon$ separation, the
ordering binaries $q_{rr'p}$ selecting exactly one direction whenever
both aircraft are assigned to $p$.

\fmlgroup{Placement compatibility}
For every $r \ne r' \in \calR$ and every $(p, p') \in \calA$:
\begin{align}
  y_{rr'pp'} &\le x_{rp},\quad
  y_{rr'pp'} \le x_{r'p'},\quad
  y_{rr'pp'} \ge x_{rp} + x_{r'p'} - 1
    \label{eq:y_link}
\end{align}
Constraints~\eqref{eq:y_link} activate the blocking interaction of the
arc $(p, p')$, $y_{rr'pp'} = 1$, exactly when both aircraft are assigned
to the corresponding positions.

\fmlgroup{Exhaustive classification of every access}
For every $(\dagger)$, the access-state indicators satisfy the following
partition constraint:
\begin{align}
  z^{-,a}_{rr'pp'} + z^{+,a}_{rr'pp'}
   + \!\!\sum_{k=1}^{N_r-1}\! b^{Ba}_{rr'pp'k}
   + \!\!\sum_{j \in \calJ_r}\! b^{Ca}_{rr'pp'j}
   &= y_{rr'pp'}
    && \label{eq:partition}
\end{align}
Constraint~\eqref{eq:partition} requires each active blocking
interaction ($y_{rr'pp'} = 1$) to select exactly one admissible access
state for $\tau^a_{r'}$, and sets all indicators to zero when the
interaction is not active ($y_{rr'pp'} = 0$).  An access that satisfies none of the four state
definitions below, because the front aircraft is mid-way through a
non-interruptible job, leaves~\eqref{eq:partition} unsatisfiable, so
the model must revise the timing or the assignment.

\fmlgroup{Before / after the front stay}
For every $(\dagger)$:
\begin{align}
  \tau^a_{r'} &\le s_r + M\bigl(1 - z^{-,a}_{rr'pp'}\bigr)
    && \label{eq:zminus}\\
  \tau^a_{r'} &\ge f_r - M\bigl(1 - z^{+,a}_{rr'pp'}\bigr)
    && \label{eq:zplus}
\end{align}
Constraint~\eqref{eq:zminus} is active only when
$z^{-,a}_{rr'pp'} = 1$, where it places the access no later than the
front aircraft's start; symmetrically, \eqref{eq:zplus} places the
access no earlier than the front's finish when $z^{+,a}_{rr'pp'} = 1$.
In both cases the front position is vacant at the access instant
(Mode~A, no manoeuvre).  The bounds are \emph{closed}: Mode~A only
requires the front position to be vacant at the access instant, so an
access that coincides with an endpoint of the stay $[s_r, f_r]$ is
admissible, as in \S\ref{subsec:access}.

\fmlgroup{Mode B --- inter-job gap}
For every $(\dagger)$ and every $k = 1, \ldots, N_r-1$:
\begin{align}
  \tau^a_{r'} &\ge f_{j^r_k} - M\bigl(1 - b^{Ba}_{rr'pp'k}\bigr)
    && \label{eq:bb_lb}\\
  \tau^a_{r'} &\le s_{j^r_{k+1}} + M\bigl(1 - b^{Ba}_{rr'pp'k}\bigr)
    && \label{eq:bb_ub}
\end{align}
Each Mode~B event consumes $\mu$ time units of the gap, so the gap
must be wide enough to accommodate all events that use it:
\begin{align}
  s_{j^r_{k+1}} - f_{j^r_k}
    &\ge
    \mu \!\!\sum_{\substack{r' \ne r\\ (p,p') \in \calA \\ a \in \{\mathrm{in}, \mathrm{out}\}}}
        \!\! b^{Ba}_{rr'pp'k}
    && \forall r \in \calR,\ k = 1, \ldots, N_r-1
    \label{eq:bb_gap}
\end{align}
The bounds~\eqref{eq:bb_lb} and~\eqref{eq:bb_ub} are active when
$b^{Ba}_{rr'pp'k} = 1$ and then confine the access to inter-job gap $k$
of the front aircraft, between the finish of $j^r_k$ and the start of
$j^r_{k+1}$; \eqref{eq:bb_gap} requires that gap to be wide enough to
absorb $\mu$ time units for every Mode~B access routed through it.

\fmlgroup{Mode C --- strictly inside an interruptible job}
For every $(\dagger)$ and every $j \in \calJ_r$:
\begin{align}
  \tau^a_{r'} &\ge s_j + \eta - M\bigl(1 - b^{Ca}_{rr'pp'j}\bigr)
    && \label{eq:bc_lb}\\
  \tau^a_{r'} &\le s_j + D_j + \delta\bigl(\kappa_j - b^{Ca}_{rr'pp'j}\bigr)
                - \eta + M\bigl(1 - b^{Ca}_{rr'pp'j}\bigr)
    && \label{eq:bc_ub}\\
  b^{Ca}_{rr'pp'j} &\le I_j
    && \label{eq:bc_interr}
\end{align}
The lower bound~\eqref{eq:bc_lb} is active when
$b^{Ca}_{rr'pp'j} = 1$ and forces the access strictly inside job $j$,
at least $\eta$ after its start.  The upper bound~\eqref{eq:bc_ub}
places the access at least $\eta$ before the job's finish, computed
\emph{without} the extension caused by the access being classified.
This exclusion is necessary because job durations depend, through
$\kappa_j$, on the classification decisions themselves: a bound written
against the full extended finish would let the extension caused by an
access make that access appear to fall within the job, and thereby
justify the interruption that generated it.
Constraint~\eqref{eq:bc_interr} forbids Mode~C on non-interruptible
jobs ($I_j = 0$).

Parameter $\eta$ represents the temporal resolution of the instance
rather than a solver tolerance; on instances with a finer time grid than
the integer days of the benchmark, it should scale with that grid rather
than be set to an arbitrarily small constant.

\fmlgroup{Counters, delay and makespan}
\begin{align}
  \kappa_j &=
    \!\!\sum_{\substack{r' \ne r\\ (p,p') \in \calA\\ a \in \{\mathrm{in}, \mathrm{out}\}}}
    \!\! b^{Ca}_{rr'pp'j}
    && \forall r \in \calR,\ j \in \calJ_r \label{eq:kappa}\\
  n &= 2 \!\!\sum_{(p,p') \in \calA}
       \!\!\sum_{\substack{r,r' \in \calR \\ r \ne r'}}
       \sum_{a}
       \biggl(
         \!\sum_{k=1}^{N_r-1}\! b^{Ba}_{rr'pp'k}
         + \!\sum_{j \in \calJ_r}\! b^{Ca}_{rr'pp'j}
       \biggr)
    && \label{eq:movements}\\
  v^D_r &\ge f_r - L_r
    && \forall r \in \calR \label{eq:delay}\\
  m &\ge f_r
    && \forall r \in \calR \label{eq:makespan}
\end{align}
Equation~\eqref{eq:kappa} aggregates the Mode~C indicators of job $j$
into its interruption counter $\kappa_j$, which feeds the duration
constraint~\eqref{eq:jobdur}; \eqref{eq:movements} counts two movements
(tow-out and tow-back) per Mode~B or Mode~C event; and
\eqref{eq:delay} and~\eqref{eq:makespan} linearise the positive delay
$v^D_r = \max(0,\, f_r - L_r)$ and the makespan under the minimisation
direction.

All disjunctions share the single constant $M$.  Pair-dependent
constants derived from the projected time windows of each aircraft pair
would strengthen the linear relaxation at no modelling cost; we retain
the uniform constant and leave this tightening to future work
(\S\ref{sec:conclusions}).

\subsection*{Objective}
The objective introduced in \S\ref{sec:problem} is
\begin{equation}
  \min \quad W^M\, m
         \;+\; W^D \sum_{r \in \calR} v^D_r
         \;+\; W^S\, n.
  \label{eq:milpobj}
\end{equation}
